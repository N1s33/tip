#include <iostream>
using namespace std;

int main() {
    double X, Y;
    cin >> X >> Y;

    int day = 1;
    double sum = X;

    while (sum < Y) {
        sum *= 1.1;
        day++;
    }

    cout << day << endl;

    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int A, B;
    cin >> A >> B;

    while (A > B) {
        if (A % 2 == 0 && A / 2 >= B) {
            cout << ":2" << endl;
            A /= 2;
        } else {
            cout << "-1" << endl;
            A--;
        }
    }

    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int N;
    cin >> N;

    int a[1000];
    int top = 0;
    int removed = 0;

    for (int i = 0; i < N; i++) {
        int x;
        cin >> x;

        if (top > 0 && a[top - 1] == x) {
            top--;
            removed += 2;
        } else {
            a[top] = x;
            top++;
        }
    }

    cout << removed << endl;

    return 0;
}
